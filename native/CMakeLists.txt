cmake_minimum_required(VERSION 3.20)
project(spindle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Spindle targets 64-bit platforms only")
endif()
option(SPINDLE_ENABLE_SIMD "Build hardware-dispatched SIMD kernels" ON)
option(SPINDLE_SHARED "Build a shared runtime" ON)
find_package(Threads REQUIRED)
add_library(spindle_options INTERFACE)
target_link_libraries(spindle_options INTERFACE Threads::Threads)
target_include_directories(spindle_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(spindle_options INTERFACE /O2 /W4 /fp:strict)
else()
  target_compile_options(spindle_options INTERFACE -O3 -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
include(CheckCXXSourceCompiles)
if(APPLE)
  find_library(ACCELERATE Accelerate REQUIRED)
  # The new CBLAS symbols require iOS 16.4. Use the legacy ABI for iOS 15.
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_compile_definitions(spindle_options INTERFACE ACCELERATE_NEW_LAPACK)
    set(CMAKE_REQUIRED_DEFINITIONS -DACCELERATE_NEW_LAPACK)
  else()
    target_compile_options(spindle_options INTERFACE -Wno-deprecated-declarations)
  endif()
  target_link_libraries(spindle_options INTERFACE ${ACCELERATE})
  set(SPINDLE_BLAS_HEADER "Accelerate/Accelerate.h")
  set(CMAKE_REQUIRED_LIBRARIES ${ACCELERATE})
  target_compile_definitions(spindle_options INTERFACE SPINDLE_BLAS_NAME="accelerate")
else()
  # FindBLAS alone does not promise the CBLAS header or C entry points.
  set(BLA_VENDOR OpenBLAS)
  set(BLA_SIZEOF_INTEGER 4)
  set(SPINDLE_OPENBLAS_ROOT "" CACHE PATH "Pinned single-threaded OpenBLAS installation")
  if(SPINDLE_OPENBLAS_ROOT)
    find_path(CBLAS_INCLUDE_DIR cblas.h PATHS "${SPINDLE_OPENBLAS_ROOT}/include" PATH_SUFFIXES openblas NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    find_library(SPINDLE_OPENBLAS_LIBRARY NAMES openblas PATHS "${SPINDLE_OPENBLAS_ROOT}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    add_library(spindle_blas INTERFACE)
    target_link_libraries(spindle_blas INTERFACE ${SPINDLE_OPENBLAS_LIBRARY})
    set(SPINDLE_BLAS_TARGET spindle_blas)
  else()
    find_package(BLAS REQUIRED)
    find_path(CBLAS_INCLUDE_DIR cblas.h PATH_SUFFIXES openblas REQUIRED)
    set(SPINDLE_BLAS_TARGET BLAS::BLAS)
  endif()
  target_include_directories(spindle_options INTERFACE ${CBLAS_INCLUDE_DIR})
  target_link_libraries(spindle_options INTERFACE ${SPINDLE_BLAS_TARGET})
  set(SPINDLE_BLAS_HEADER "cblas.h")
  set(CMAKE_REQUIRED_INCLUDES ${CBLAS_INCLUDE_DIR})
  # A check's try_compile project cannot see a non-imported interface target.
  if(SPINDLE_OPENBLAS_ROOT)
    set(CMAKE_REQUIRED_LIBRARIES ${SPINDLE_OPENBLAS_LIBRARY} Threads::Threads)
  else()
    set(CMAKE_REQUIRED_LIBRARIES BLAS::BLAS)
  endif()
  target_compile_definitions(spindle_options INTERFACE SPINDLE_BLAS_NAME="openblas")
endif()
check_cxx_source_compiles("#include <${SPINDLE_BLAS_HEADER}>
#include <type_traits>
using CblasGemm = void (*)(CBLAS_ORDER,CBLAS_TRANSPOSE,CBLAS_TRANSPOSE,int,int,int,float,const float*,int,const float*,int,float,float*,int);
static_assert(std::is_same<decltype(&cblas_sgemm),CblasGemm>::value);
int main() { float x[4]={}; cblas_sgemm(CblasRowMajor,CblasNoTrans,CblasNoTrans,1,1,1,1,x,1,x,1,0,x,1); cblas_sgemv(CblasRowMajor,CblasNoTrans,1,1,1,x,1,x,1,0,x,1); }" SPINDLE_HAS_LP64_CBLAS)
if(NOT SPINDLE_HAS_LP64_CBLAS)
  message(FATAL_ERROR "Spindle requires LP64 CBLAS headers and sgemm/sgemv symbols")
endif()
set(SPINDLE_KERNELS kernels.cpp kernels_scalar.cpp)
set(SPINDLE_ARCH "${CMAKE_SYSTEM_PROCESSOR}")
if(APPLE AND CMAKE_OSX_ARCHITECTURES)
  set(SPINDLE_ARCH "${CMAKE_OSX_ARCHITECTURES}")
endif()
set(SPINDLE_EXPECT_ARCH "" CACHE STRING "Expected target: x86_64 or arm64")
if(SPINDLE_EXPECT_ARCH STREQUAL "x86_64" AND NOT SPINDLE_ARCH MATCHES "^(x86_64|AMD64|amd64)$")
  message(FATAL_ERROR "This preset requires an x86-64 toolchain")
elseif(SPINDLE_EXPECT_ARCH STREQUAL "arm64" AND NOT SPINDLE_ARCH MATCHES "^(arm64|aarch64|ARM64)$")
  message(FATAL_ERROR "This preset requires an ARM64 toolchain")
endif()
if(SPINDLE_ENABLE_SIMD)
  if(SPINDLE_ARCH MATCHES "^(x86_64|AMD64|amd64)$")
    list(APPEND SPINDLE_KERNELS kernels_sse2.cpp kernels_avx2.cpp)
    target_compile_definitions(spindle_options INTERFACE SPINDLE_HAVE_X86)
    if(MSVC)
      set_source_files_properties(kernels_avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    else()
      set_source_files_properties(kernels_sse2.cpp PROPERTIES COMPILE_OPTIONS "-msse2;-mno-avx")
      set_source_files_properties(kernels_avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mno-fma")
    endif()
  elseif(SPINDLE_ARCH MATCHES "^(arm64|aarch64|ARM64)$")
    list(APPEND SPINDLE_KERNELS kernels_neon.cpp)
    target_compile_definitions(spindle_options INTERFACE SPINDLE_HAVE_NEON)
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("-march=armv8.2-a+dotprod" SPINDLE_CAN_BUILD_DOTPROD)
    if(SPINDLE_CAN_BUILD_DOTPROD)
      list(APPEND SPINDLE_KERNELS kernels_neon_dot.cpp)
      set_source_files_properties(kernels_neon_dot.cpp PROPERTIES COMPILE_OPTIONS "-march=armv8.2-a+dotprod")
      target_compile_definitions(spindle_options INTERFACE SPINDLE_HAVE_NEON_DOT)
    endif()
  endif()
endif()
# The diagnostic baseline must not be silently vectorized by the compiler.
if(MSVC)
  set_source_files_properties(kernels_scalar.cpp PROPERTIES COMPILE_OPTIONS "/Qvec-")
elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  set_source_files_properties(kernels_scalar.cpp PROPERTIES COMPILE_OPTIONS "-fno-vectorize;-fno-slp-vectorize")
else()
  set_source_files_properties(kernels_scalar.cpp PROPERTIES COMPILE_OPTIONS "-fno-tree-vectorize")
endif()
set(SPINDLE_SOURCES archive.cpp math.cpp cq4.cpp workers.cpp decoder.cpp confidence.cpp generation.cpp grammar.cpp api.cpp ${SPINDLE_KERNELS})
if(SPINDLE_SHARED)
  add_library(spindle SHARED ${SPINDLE_SOURCES})
else()
  add_library(spindle STATIC ${SPINDLE_SOURCES})
  target_compile_definitions(spindle PUBLIC SPINDLE_STATIC)
endif()
target_link_libraries(spindle PRIVATE spindle_options)
target_include_directories(spindle PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(spindle PRIVATE SPINDLE_BUILD)
set_target_properties(spindle PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
  INTERPROCEDURAL_OPTIMIZATION FALSE POSITION_INDEPENDENT_CODE YES)
if(ANDROID)
  target_link_options(spindle PRIVATE "-Wl,-z,max-page-size=16384")
endif()
install(TARGETS spindle LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(FILES spindle.h DESTINATION include)
include(CTest)
if(BUILD_TESTING)
  foreach(name cq4 attention kernels integer)
    add_executable(test_${name} test_${name}.cpp archive.cpp math.cpp cq4.cpp workers.cpp ${SPINDLE_KERNELS})
    target_link_libraries(test_${name} PRIVATE spindle_options)
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
  set_tests_properties(cq4 PROPERTIES ENVIRONMENT "SPINDLE_EXACT_PREFILL=1")
  set_tests_properties(integer PROPERTIES ENVIRONMENT "SPINDLE_EXACT_PREFILL=0")
  set(SPINDLE_TEST_KERNELS scalar)
  if(SPINDLE_ARCH MATCHES "^(x86_64|AMD64|amd64)$" AND SPINDLE_ENABLE_SIMD)
    list(APPEND SPINDLE_TEST_KERNELS sse2 avx2)
  elseif(SPINDLE_ARCH MATCHES "^(arm64|aarch64|ARM64)$" AND SPINDLE_ENABLE_SIMD)
    list(APPEND SPINDLE_TEST_KERNELS neon)
    if(SPINDLE_CAN_BUILD_DOTPROD)
      list(APPEND SPINDLE_TEST_KERNELS neon-dot)
    endif()
  endif()
  foreach(kernel IN LISTS SPINDLE_TEST_KERNELS)
    foreach(name cq4 attention kernels integer)
      add_test(NAME ${name}_${kernel} COMMAND test_${name})
      set_tests_properties(${name}_${kernel} PROPERTIES
        ENVIRONMENT "SPINDLE_KERNEL=${kernel};SPINDLE_EXACT_PREFILL=1" SKIP_RETURN_CODE 77)
      if(name STREQUAL "integer")
        set_tests_properties(${name}_${kernel} PROPERTIES ENVIRONMENT "SPINDLE_KERNEL=${kernel};SPINDLE_EXACT_PREFILL=0")
      endif()
    endforeach()
  endforeach()
endif()
